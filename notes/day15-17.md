# ping_result.py：遍历设备清单，打印巡检结论（纯逻辑版，真 ping 在 Day 17）
devices = ["192.168.1.1", "192.168.1.2", "192.168.1.3"]

for i in range(3):                    # range(3) = 0,1,2
    print("第", i + 1, "轮")

status = "up"
if status == "up":                    # if 判断（结尾冒号别丢）
    print("接口正常")
elif status == "down":
    print("接口故障")
else:
    print("未知状态")

count = 0
while count < 3:                      # while 循环
    print("重试第", count + 1, "次")

Day 17：函数 + 用 Python 重写 sys_check.sh（本周高潮）
Python
# sys_check.py
import subprocess                      # 调用系统命令的标准库

def run(cmd):                          # 定义函数：执行命令并返回输出文本
    result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    return result.stdout

def main():
    print("===== 系统巡检报告（Python 版）=====")
    print("--- 负载 ---");   print(run("uptime"))
    print("--- 内存 ---");   print(run("free -h"))
    print("--- 磁盘 ---");   print(run("df -h /"))
    print("--- 监听端口 ---"); print(run("ss -tln"))

if __name__ == "__main__":             # Python 脚本的"主入口"固定写法
    main()
跑通后对比 Shell 版写进笔记：哪个更短？字符串处理哪个顺手？（体会：Shell 擅长拼命令，Python 擅长处理数据和逻辑——第 2 个月开始你会越来越多用后者。）

2.验收
(1)f-string 怎么写？f"设备{name}" 里花括号的作用？
答:字符串前加f，{}里的变量/表达式会被替换成实际值

(2)range(3) 产生哪几个数？
答:0、1、2(从0开始，不含终点3)

(3)Python 靠什么划分代码块？（和 Shell 的 done/fi 对比）
答:缩进(4个空格)--同一代码块缩进必须一致；Shell用done/fi，Python用缩进

(4)在 Python 里执行系统命令用哪个模块哪个函数？
答:subprocess模块的subprocess.run() （配capture_output=True,text=True拿输出）

(5)if __name__ == "__main__": 是干什么的？
答:只有"直接运行这个文件"时才执行下面的代码；被别的脚本 import 时不执行——模块化的基础写法
